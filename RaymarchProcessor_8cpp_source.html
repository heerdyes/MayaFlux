62uint32_t RaymarchProcessor::auto_steps() const
63{
64 constexpr uint32_t k_cap = 512;
65
66 const glm::vec3 cell = m_lattice.cell_size();
67 const float step = std::min({ cell.x, cell.y, cell.z }) * std::max(m_params.step_scale, 1e-3F);
68 const float span = glm::length(m_lattice.bounds.max - m_lattice.bounds.min);
69
70 return std::min(k_cap, static_cast<uint32_t>(std::ceil(span / step)) + 1U);
71}

147void RaymarchProcessor::stage_descriptor(
148 const std::shared_ptr<VKBuffer>& buffer, vk::Buffer handle)
149{
150 auto& bindings = buffer->get_pipeline_context().descriptor_buffer_bindings;
151
152 auto it = std::ranges::find_if(bindings, [this](const auto& entry) {
153 return entry.set == m_set && entry.binding == m_binding;

156 if (it == bindings.end()) {
157 bindings.push_back(Portal::Graphics::DescriptorBindingInfo {

162 .name = k_binding_name,
163 .count = 1,
164 });
165 return;
166 }

168 it->buffer_info.buffer = handle;
169 it->buffer_info.offset = 0;

173void RaymarchProcessor::stage_params(const std::shared_ptr<VKBuffer>& buffer)
174{
175 auto& fragments = buffer->get_pipeline_context().push_constant_bindings;
176
177 const auto* bytes = reinterpret_cast<const uint8_t*>(&m_params);
178
179 auto it = std::ranges::find_if(fragments, [](const auto& entry) {
180 return entry.name == k_fragment_name;
181 });
182
183 if (it == fragments.end()) {
184 fragments.push_back(Portal::Graphics::PushConstantBindingInfo {
185 .offset = 0,
186 .data = std::vector<uint8_t>(bytes, bytes + sizeof(MarchParams)),
187 .name = k_fragment_name,
188 });
189 return;
190 }
191
192 it->offset = 0;
193 it->data.assign(bytes, bytes + sizeof(MarchParams));
194}
